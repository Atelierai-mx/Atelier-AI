/* ATELIER AI 7.1 Correccion de catalogo y presupuesto. Pegar antes de en index.html. */ // Reconocer los nombres de prendas que utiliza la IA. function categoriaAtelier(valor) { const v = normalize(valor) .replace(/\b(de|para|tipo|unos|unas|un|una|el|la|los|las)\b/g, ' ') .replace(/\s+/g, ' ') .trim(); if (/\b(polo|playera|camiseta|t-shirt|sueter|sudadera)\b/.test(v)) { return 'no-catalogada'; } if (/\b(camisa|camisas)\b/.test(v)) { return 'camisa'; } if (/\b(pantalon|pantalones|trousers)\b/.test(v)) { return 'pantalon'; } if (/\b(mocasin|mocasines|loafer|loafers|zapato|zapatos|calzado)\b/.test(v)) { return 'zapatos'; } return v; } // Revisar detalles sin rechazar automaticamente la prenda. function detallesCompatibles(s, p) { const d = normalize(s.detalles); if ( !d || /^(null|ninguno|ninguna|sin detalles|no especificado)$/.test(d) ) { return true; } const ficha = normalize([p.name, p.fit].join(' ')); const requisitos = [ [/\blino\b/, /\blino\b/], [/\bpiel\b|\bcuero\b/, /\bpiel\b|\bcuero\b/], [/\b100\s*%?\s*algodon\b/, /\b100\s*%?\s*algodon\b/], [/\boversize\b/, /\boversize\b/], [/\bmanga corta\b/, /\bmanga corta\b/], [/\bmanga larga\b/, /\bmanga larga\b/] ]; for (const [pedido, verificacion] of requisitos) { if (pedido.test(d) && !verificacion.test(ficha)) { return false; } } if ( /\b(oxford|seda|terciopelo|impermeable)\b/.test(d) && !/\b(oxford|seda|terciopelo|impermeable)\b/.test(ficha) ) { return false; } return true; } // Mejorar la coincidencia entre la solicitud y el catalogo. strictMatch = function(s, p) { if ( categoriaAtelier(s.tipo) !== categoriaAtelier(productType(p)) ) { return false; } const c = normalize(s.color); if ( c && c !== 'null' && c !== 'no especificado' && c !== normalize(p.color) ) { return false; } const f = normalize(s.corte); const ficha = normalize([p.fit, p.name].join(' ')); if (f && f !== 'null' && f !== 'no especificado') { if ( /\b(slim|entallado)\b/.test(f) && !/\bslim\b/.test(ficha) ) { return false; } if ( /\b(holgado|loose|relaxed)\b/.test(f) && !/\b(holgado|loose|relaxed)\b/.test(ficha) ) { return false; } if ( /\bregular\b/.test(f) && !/\bregular\b/.test(ficha) ) { return false; } if ( /\b(mocasin|mocasines|loafer|loafers)\b/.test(f) && !/\b(mocasin|mocasines|loafer|loafers)\b/.test(ficha) ) { return false; } } return detallesCompatibles(s, p); }; // Seleccionar la mayor cantidad posible de prendas // sin superar el presupuesto indicado. reconcile = function(items) { const limite = budget(); const out = items.map(s => ({ ...s, product: null })); let estados = [{ total: 0, elegidos: [], cantidad: 0, conservados: 0 }]; for (let i = 0; i < out.length; i++) { const s = out[i]; const opciones = choices(s); const anterior = items[i]?.product; if ( anterior && opciones.some(p => p.id === anterior.id) ) { opciones.sort( (a, b) => (b.id === anterior.id) - (a.id === anterior.id) || a.price - b.price ); } const siguientes = []; for (const e of estados) { siguientes.push({ ...e, elegidos: [...e.elegidos, null] }); for (const p of opciones) { if ( e.total + p.price <= limite && !e.elegidos.some(x => x?.id === p.id) ) { siguientes.push({ total: e.total + p.price, elegidos: [...e.elegidos, p], cantidad: e.cantidad + 1, conservados: e.conservados + (anterior?.id === p.id ? 1 : 0) }); } } } siguientes.sort( (a, b) => b.cantidad - a.cantidad || b.conservados - a.conservados || a.total - b.total ); estados = siguientes.slice(0, 2000); } const ganador = estados[0]; out.forEach((s, i) => { s.product = ganador?.elegidos[i] || null; }); return out; }; // Recalcular la seleccion cuando cambia el presupuesto. let ultimoPresupuesto = budget(); $('budget').addEventListener('input', () => { if (!outfit.length || busy) { return; } const nuevo = budget(); if (!nuevo || nuevo === ultimoPresupuesto) { return; } ultimoPresupuesto = nuevo; outfit = reconcile(outfit); render(); status( 'Presupuesto ajustado a ' + money(nuevo) + '. Los precios son referencias y el inventario no esta verificado.' ); });
